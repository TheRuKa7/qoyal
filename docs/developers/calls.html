API reference

Calls

A call is one conversation with one person. Place single calls from your own events, such as a new order, and read back transcripts, outcomes and recordings for any call.

POST/v1/calls

Place a call

Dials one number with an agent. Values in context fill the agent’s inputs for this call only.

Body

agentstringrequired

Agent id.

tostringrequired

Number to call. E.164, or 10 digits for India (+91 is added).

contextobject

Input values by key. Missing required inputs are passed as “(not provided)”.

voicestring

Override the agent’s voice for this call.

metadataobject

Your own references, returned on the call and in webhooks.

Request
curl -X POST "$ECHO_API/v1/calls" \
  -H "Authorization: Bearer $ECHO_API_KEY" \
  -H "Content-Type: application/json" \
  -d '{
    "agent": "dispatch-followup",
    "to": "+919876543210",
    "context": {
      "supplier_name": "Mehta Packaging",
      "po_number": "4471",
      "open_qty": 14
    },
    "metadata": {
      "erp_line_id": "L-44713"
    }
  }'
import os, requests

API = os.environ["ECHO_API"]
HEADERS = {"Authorization": f"Bearer {os.environ['ECHO_API_KEY']}"}

res = requests.post(
    f"{API}/v1/calls",
    headers=HEADERS,
    json={
  "agent": "dispatch-followup",
  "to": "+919876543210",
  "context": {
    "supplier_name": "Mehta Packaging",
    "po_number": "4471",
    "open_qty": 14
  },
  "metadata": {
    "erp_line_id": "L-44713"
  }
},
)
res.raise_for_status()
print(res.json())
const API = process.env.ECHO_API;
const KEY = process.env.ECHO_API_KEY;

const res = await fetch(`${API}/v1/calls`, {
  method: "POST",
  headers: { Authorization: `Bearer ${KEY}`, "Content-Type": "application/json" },
  body: JSON.stringify({
    "agent": "dispatch-followup",
    "to": "+919876543210",
    "context": {
      "supplier_name": "Mehta Packaging",
      "po_number": "4471",
      "open_qty": 14
    },
    "metadata": {
      "erp_line_id": "L-44713"
    }
  }),
}});
if (!res.ok) throw new Error(`Echo API ${res.status}`);
console.log(await res.json());

GET/v1/calls/{call_id}

Retrieve a call

Returns the call with its context, reported outputs, transcript, recording link and analysis when it exists.

Path parameters

call_idstringrequired

Notestatus is queued, in_progress or completed. A completed call with turns: 0 was answered but had no audio.
Request
curl "$ECHO_API/v1/calls/call_8f2k1q" \
  -H "Authorization: Bearer $ECHO_API_KEY"
import os, requests

API = os.environ["ECHO_API"]
HEADERS = {"Authorization": f"Bearer {os.environ['ECHO_API_KEY']}"}

res = requests.get(
    f"{API}/v1/calls/call_8f2k1q",
    headers=HEADERS,
)
res.raise_for_status()
print(res.json())

const KEY = process.env.ECHO_API_KEY;

console.log(await res.json());

GET/v1/calls

List calls

The call log, newest first, with the same filters as the dashboard.

Query parameters

agentstring

campaign_idstring

qstring

Search by name or number.

sincetimestamp

ISO 8601.

untiltimestamp

min_durationinteger

Seconds.

max_durationinteger

Seconds.

limitinteger

1 to 100.

cursorstring

Request
curl "$ECHO_API/v1/calls?agent=dispatch-followup&since=2026-09-18T00:00:00Z&limit=50" \
  -H "Authorization: Bearer $ECHO_API_KEY"
import os, requests

GET/v1/calls/{call_id}/recording

Download a recording

The call audio.

Path parameters

call_idstringrequired

Request
curl "$ECHO_API/v1/calls/call_8f2k1q/recording" \
  -H "Authorization: Bearer $ECHO_API_KEY" \
  -o call.wav
import os, requests

API = os.environ["ECHO_API"]
HEADERS = {"Authorization": f"Bearer {os.environ['ECHO_API_KEY']}"}

res = requests.get(
    f"{API}/v1/calls/call_8f2k1q/recording",
    headers=HEADERS,
)
res.raise_for_status()
open("call.wav", "wb").write(res.content)
import fs from "node:fs";
const API = process.env.ECHO_API;
const KEY = process.env.ECHO_API_KEY;

const res = await fetch(`${API}/v1/calls/call_8f2k1q/recording`, {
  headers: { Authorization: `Bearer ${KEY}` },
}});
if (!res.ok) throw new Error(`Echo API ${res.status}`);
await fs.promises.writeFile("call.wav", Buffer.from(await res.arrayBuffer()));

GET/v1/calls/export

Export the call log

The filtered call log as a spreadsheet. Takes the same filters as List calls.

Request
curl "$ECHO_API/v1/calls/export?since=2026-09-01T00:00:00Z" \
  -H "Authorization: Bearer $ECHO_API_KEY" \
  -o calls.xlsx
import os, requests

API = os.environ["ECHO_API"]
HEADERS = {"Authorization": f"Bearer {os.environ['ECHO_API_KEY']}"}

res = requests.get(
    f"{API}/v1/calls/export",
    headers=HEADERS,
    params={"since": "2026-09-01T00:00:00Z"},
)
res.raise_for_status()
open("calls.xlsx", "wb").write(res.content)
import fs from "node:fs";
const API = process.env.ECHO_API;
const KEY = process.env.ECHO_API_KEY;

const res = await fetch(`${API}/v1/calls/export?since=2026-09-01T00:00:00Z`, {
  headers: { Authorization: `Bearer ${KEY}` },
}});
if (!res.ok) throw new Error(`Echo API ${res.status}`);
await fs.promises.writeFile("calls.xlsx", Buffer.from(await res.arrayBuffer()));
Response · 200
Binary .xlsx file
