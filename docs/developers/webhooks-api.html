API reference

Webhooks API

Get each outcome the moment it lands instead of polling. Register an endpoint, then verify every delivery.

PreviewPart of the developer preview. Fields may change before general availability.
POST/v1/webhooks

Register an endpoint

Subscribes a URL to events. The response includes the signing secret, shown once.

Body

urlstringrequired

HTTPS endpoint.

eventsarrayrequired

Any of call.completed, call.analyzed, contact.captured, campaign.completed.

descriptionstring

Request
curl -X POST "$ECHO_API/v1/webhooks" \
  -H "Authorization: Bearer $ECHO_API_KEY" \
  -H "Content-Type: application/json" \
  -d '{
    "url": "https://erp.example.com/echo/outcomes",
    "events": [
      "contact.captured",
      "campaign.completed"
    ]
  }'
import os, requests

API = os.environ["ECHO_API"]
HEADERS = {"Authorization": f"Bearer {os.environ['ECHO_API_KEY']}"}

res = requests.post(
    f"{API}/v1/webhooks",
    headers=HEADERS,
    json={
  "url": "https://erp.example.com/echo/outcomes",
  "events": [
    "contact.captured",
    "campaign.completed"
  ]
},
)
res.raise_for_status()
print(res.json())
const API = process.env.ECHO_API;
const KEY = process.env.ECHO_API_KEY;

const res = await fetch(`${API}/v1/webhooks`, {
  method: "POST",
  headers: { Authorization: `Bearer ${KEY}`, "Content-Type": "application/json" },
  body: JSON.stringify({
    "url": "https://erp.example.com/echo/outcomes",
    "events": [
      "contact.captured",
      "campaign.completed"
    ]
  }),
}});
if (!res.ok) throw new Error(`Echo API ${res.status}`);
console.log(await res.json());
Response · 201
{
  "id": "wh_31k",
  "url": "https://erp.example.com/echo/outcomes",
  "events": [
    "contact.captured",
    "campaign.completed"
  ],
  "secret": "whsec_...",
  "created_at": "2026-09-25T09:00:00Z"
}
GET/v1/webhooks

List endpoints

Your registered endpoints. Secrets are not returned.

Request
curl "$ECHO_API/v1/webhooks" \
  -H "Authorization: Bearer $ECHO_API_KEY"
import os, requests

API = os.environ["ECHO_API"]
HEADERS = {"Authorization": f"Bearer {os.environ['ECHO_API_KEY']}"}

res = requests.get(
    f"{API}/v1/webhooks",
    headers=HEADERS,
)
res.raise_for_status()
print(res.json())
const API = process.env.ECHO_API;
const KEY = process.env.ECHO_API_KEY;

const res = await fetch(`${API}/v1/webhooks`, {
  headers: { Authorization: `Bearer ${KEY}` },
}});
if (!res.ok) throw new Error(`Echo API ${res.status}`);
console.log(await res.json());
Response · 200
{
  "data": [
    {
      "id": "wh_31k",
      "url": "https://erp.example.com/echo/outcomes",
      "events": [
        "contact.captured",
        "campaign.completed"
      ]
    }
  ]
}

DELETE/v1/webhooks/{webhook_id}

Remove an endpoint

Stops deliveries to this URL.

Path parameters

webhook_idstringrequired

Request
curl -X DELETE "$ECHO_API/v1/webhooks/wh_31k" \
  -H "Authorization: Bearer $ECHO_API_KEY"
import os, requests

API = os.environ["ECHO_API"]
HEADERS = {"Authorization": f"Bearer {os.environ['ECHO_API_KEY']}"}

res = requests.delete(
    f"{API}/v1/webhooks/wh_31k",
    headers=HEADERS,
)
res.raise_for_status()
print(res.json())
const API = process.env.ECHO_API;
const KEY = process.env.ECHO_API_KEY;

const res = await fetch(`${API}/v1/webhooks/wh_31k`, {
  method: "DELETE",
  headers: { Authorization: `Bearer ${KEY}` },
}});
if (!res.ok) throw new Error(`Echo API ${res.status}`);
console.log(await res.json());
Response · 200
{
  "id": "wh_31k",
  "deleted": true
}
